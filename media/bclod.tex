		{\LARGE Base Loaded antenna capacitance} \bigbreak
		\bigbreak
	
	$ C = \frac{55.78h}{[(den1) \times (den2)]} $ \medbreak
	\end{center}

	\hspace{\wh mm} where
	
	\halign
	{\quad \hspace{\bump mm} # \hfil & # \hfil \cr
 		& $ (den1) = \ln(\frac{h}{r}) - 1 $ \cr
		& $ (den2) = [1 - (\frac{fh}{75}) ^ 2] $ \cr
		& ln = natural logarathm \cr
		& \enspace $ C $ is capacitance, $ f $ is frequency, $ h $ is height \cr
		& without coil in meters, and $ r $ is wire radius in \cr
		& meters. \cr
	}
