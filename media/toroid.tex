	\bigbreak
	
	$ C_{(d_1, d_2)} = \ddfrac{ 2.8 \left( 1.2781 - \ddfrac{d_2}{d_1} \right) \sqrt { \ddfrac{\pi}{4} (d_1 - d_2)d_2} }{25.4} $
	\bigbreak
	\bigbreak

	$ C_{(x)} = ( a \cdot 1.2781 - ax ) \sqrt { x - x ^ 2 } $
	\break
	\break

	$ a = \ddfrac { 2.8 \sqrt {\pi} d_1 }{ 2 \cdot 25.4 } $
	\break
		
	$ x = \ddfrac{ d_2  }{ d_1 } = 0.2909 $
